! ============================================================
! Device: SW1 — HQ Core Switch
! Role: Core switching, trunking, VLAN routing handoff to R1
! ============================================================

hostname SW1_HQ_Core_Switch

! ── SSH ──────────────────────────────────────────────────────
ip domain-name corp.lab
ip ssh version 2
username admin secret Admin@123
crypto key generate rsa modulus 2048

! ── DHCP Snooping ────────────────────────────────────────────
ip dhcp snooping
ip dhcp snooping vlan 10-40
no ip dhcp snooping information option

! ── STP ──────────────────────────────────────────────────────
spanning-tree mode rapid-pvst

! ── Interfaces ───────────────────────────────────────────────
! Trunk toward R1 — trusted for DHCP snooping
interface FastEthernet0/1
 switchport mode trunk
 ip dhcp snooping trust

! Trunk toward SW2
interface FastEthernet0/2
 switchport mode trunk

! Access port — Server0 (VLAN 40, static IP 192.168.40.2)
interface FastEthernet0/3
 switchport mode access
 switchport access vlan 40
 switchport port-security
 switchport port-security maximum 1
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 ip dhcp snooping limit rate 15
 spanning-tree portfast

! ── Management Interface ──────────────────────────────────────
interface Vlan40
 ip address 192.168.40.3 255.255.255.248
 no shutdown

ip default-gateway 192.168.40.1

! ── VLANs ────────────────────────────────────────────────────
vlan 10
 name Sales
vlan 20
 name IT
vlan 30
 name HR
vlan 40
 name Management

! ── Line Configuration ────────────────────────────────────────
line con 0
 password Cisco@123
 login

line vty 0 4
 login local
 transport input ssh
